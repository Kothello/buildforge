     ```ts

       | "LEFT_VM"
       | "CALLBACK_3_DAYS"
       | "CALLBACK_7_DAYS"
       | "NOT_INTERESTED";

     interface DispositionTemplate {
       id: DispositionTemplateId;
       label: string;
       dispositionText: string;     // value to put into the disposition field

       callbackOffsetDays?: number; // if set, default next callback date = now + this many days
     ```
